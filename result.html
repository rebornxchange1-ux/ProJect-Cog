function renderDoc(raw) { let text = raw .replace(/\\n/g, '\n') .replace(/\\"/g, '"') .replace(/\\t/g, ' '); const lines = text.split('\n'); let html = ''; let lastRenderedHeading = ''; let inParagraph = false; let inCoverPage = true; // true until we hit DECLARATION let inTOC = false; // true between TABLE OF CONTENTS and next chapter heading function closeParagraph() { if (inParagraph) { html += '

'; inParagraph = false; } } function cleanLine(line) { return line .replace(/\*\*(.+?)\*\*/g, '$1') .replace(/\*(.+?)\*/g, '$1') .replace(/\*/g, '') .replace(/={3,}/g, '') .replace(/#{1,6}\s*/g, '') .replace(/^[-–]{3,}$/, '') .trim(); } for (let i = 0; i < lines.length; i++) { const raw_line = lines[i]; const line = raw_line.trim(); if (!line) { closeParagraph(); html += '
'; continue; } if (/^(COVER PAGE|FREE PREVIEW|plaintext|```plaintext|```)$/i.test(line)) continue; if (/DEFENSE PREPARATION NOTICE/i.test(line) && currentType !== 'final_year_project') continue; if (/=+/.test(line) && line.replace(/=/g,'').trim() === '') continue; if (/^CHAPTER ONE:\s*INTRODUCTION$/i.test(line) && (currentType === 'seminar' || currentType === 'term_paper')) continue; if (/DEFENSE PREPARATION NOTICE/i.test(line) && currentType === 'final_year_project') { closeParagraph(); html += `
🎓 DEFENSE PREPARATION NOTICE

Congratulations on completing your Final Year Project! Your next step is to prepare for your project defense. ProjectCog has a Mock Defense Simulator that will help you practice answering panel questions before your real defense day.

`; while (i + 1 < lines.length && lines[i+1].trim() !== '' && !/^(CHAPTER|SECTION|\*\*CHAPTER|\*\*SECTION|REFERENCES)/.test(lines[i+1].trim())) { i++; } continue; } if (/^\[NEW PAGE\]$/i.test(line) || /^---+$/.test(line) || /^===+$/.test(line)) { closeParagraph(); html += '
'; continue; } // ── COVER PAGE: everything before DECLARATION gets its own centered line ── if (inCoverPage) { if (/^DECLARATION$/i.test(line)) { inCoverPage = false; // fall through to normal heading handling below } else { closeParagraph(); const cleaned = cleanLine(line); if (cleaned) { html += `

${cleaned}

`; } continue; } } // ── TABLE OF CONTENTS: keep each entry on its own line ── if (/^TABLE OF CONTENTS$/i.test(line)) { closeParagraph(); inTOC = true; html += `

${cleanLine(line)}

`; continue; } if (inTOC) { // TOC ends when we hit the first real chapter/heading line if (/^(CHAPTER ONE|CHAPTER 1|SECTION ONE)/i.test(line)) { inTOC = false; // fall through to normal handling } else { closeParagraph(); // Split entries that may be jammed on one line separated by 2+ spaces or known keywords const entries = cleanLine(line).split(/(?=Chapter\s|Section\s|\d+\.\d+\s)/i).map(e => e.trim()).filter(Boolean); entries.forEach(entry => { html += `

${entry}

`; }); continue; } } if (line.startsWith('# ')) { closeParagraph(); html += `

${cleanLine(line.slice(2))}

`; continue; } if (line.startsWith('## ')) { closeParagraph(); html += `

${cleanLine(line.slice(3))}

`; continue; } if (line.startsWith('### ')) { closeParagraph(); html += `

${cleanLine(line.slice(4))}

`; continue; } if (/^[*=\-_]{3,}$/.test(line)) { closeParagraph(); html += '
'; continue; } if (line === line.toUpperCase() && line.length < 80 && !/[.,:;?!]/.test(line) && line.length > 2) { closeParagraph(); const KNOWN_HEADINGS = ['DEDICATION','ACKNOWLEDGEMENT','ABSTRACT','DECLARATION','CERTIFICATION','TABLE OF CONTENTS','CHAPTER ONE','CHAPTER TWO','CHAPTER THREE','CHAPTER FOUR','CHAPTER FIVE','INTRODUCTION']; const stripped = cleanLine(line).replace(/<[^>]+>/g,'').toUpperCase().trim(); if (KNOWN_HEADINGS.includes(stripped)) { if (lastRenderedHeading === stripped) continue; lastRenderedHeading = stripped; } html += `

${cleanLine(line)}

`; continue; } if (/^_{3,}/.test(line) || /Signed:\s*_{2,}/.test(line) || /Date:\s*_{2,}/.test(line)) { closeParagraph(); html += `

${line}

`; continue; } const cleaned = cleanLine(line); if (!cleaned) continue; if (!inParagraph) { html += `

`; inParagraph = true; } else { html += ' '; } html += cleaned; } closeParagraph(); return html; }
